Private Sub DVBCMClient1_OnBatchStateChanged(ByVal bstrCampaignID _
        As String, ByVal bstrBatchID As String, ByVal dvbBatchState _

End Sub

Parameter Name

Purpose

The parent campaign of the batch that changed.

bstrBatchID

The user defined Batch ID of the batch.

dvbBatchState
